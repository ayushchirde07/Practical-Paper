Linear Search

Checks each element one by one until the target is found.

Time: Best O(1), worst O(n)Auxiliary space: O(1)
C++ source code
#include <iostream>
using namespace std;

int main() {
    int a[] = {10, 20, 30, 40, 50};
    int n = 5, key = 30;

    for (int i = 0; i < n; i++) {
        if (a[i] == key) {
            cout << "Element found at position " << i + 1;
            return 0;
        }
    }

    cout << "Element not found";
    return 0;
}

Sample output

Element found at position 3

Binary Search

Searches a sorted array by repeatedly halving the search range.

Time: O(log n)Auxiliary space: O(1)
C++ source code
#include <iostream>
using namespace std;

int main() {
    int a[] = {10, 20, 30, 40, 50};
    int n = 5, key = 40;
    int low = 0, high = n - 1;

    while (low <= high) {
        int mid = low + (high - low) / 2;

        if (a[mid] == key) {
            cout << "Element found at position " << mid + 1;
            return 0;
        } else if (a[mid] < key) {
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }

    cout << "Element not found";
    return 0;
}

Sample output

Element found at position 4

Important: Binary Search requires the array to be sorted.

Bubble Sort

Compares adjacent elements and swaps them when they are in the wrong order.

Time: O(n²) for this versionAuxiliary space: O(1)
C++ source code
#include <iostream>
using namespace std;

int main() {
    int a[] = {5, 3, 8, 1, 2};
    int n = 5;

    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (a[j] > a[j + 1]) {
                int temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    for (int i = 0; i < n; i++)
        cout << a[i] << " ";

    return 0;
}

Sample output

1 2 3 5 8

Selection Sort

Selects the smallest element from the unsorted part and places it next.

Time: O(n²)Auxiliary space: O(1)
C++ source code
#include <iostream>
using namespace std;

int main() {
    int a[] = {5, 3, 8, 1, 2};
    int n = 5;

    for (int i = 0; i < n - 1; i++) {
        int minIndex = i;

        for (int j = i + 1; j < n; j++) {
            if (a[j] < a[minIndex])
                minIndex = j;
        }

        int temp = a[i];
        a[i] = a[minIndex];
        a[minIndex] = temp;
    }

    for (int i = 0; i < n; i++)
        cout << a[i] << " ";

    return 0;
}

Sample output

1 2 3 5 8

Merge Sort

Divides the array, sorts each half, then merges the halves.

Time: O(n log n)Auxiliary space: O(n)
C++ source code
#include <iostream>
using namespace std;

void merge(int a[], int l, int m, int r) {
    int temp[100];
    int i = l, j = m + 1, k = 0;

    while (i <= m && j <= r) {
        if (a[i] <= a[j])
            temp[k++] = a[i++];
        else
            temp[k++] = a[j++];
    }

    while (i <= m) temp[k++] = a[i++];
    while (j <= r) temp[k++] = a[j++];

    for (i = l, k = 0; i <= r; i++, k++)
        a[i] = temp[k];
}

void mergeSort(int a[], int l, int r) {
    if (l < r) {
        int m = l + (r - l) / 2;
        mergeSort(a, l, m);
        mergeSort(a, m + 1, r);
        merge(a, l, m, r);
    }
}

int main() {
    int a[] = {9, 4, 7, 2, 6, 1};
    int n = 6;

    mergeSort(a, 0, n - 1);

    for (int i = 0; i < n; i++)
        cout << a[i] << " ";

    return 0;
}

Sample output

1 2 4 6 7 9

This teaching version uses a temporary array of size 100, so keep the input size within that limit.

Heap Sort

Builds a Max Heap and repeatedly moves the maximum element to the end.

Time: O(n log n)Auxiliary space: O(log n) recursion stack in this implementation
C++ source code
#include <iostream>
#include <algorithm>
using namespace std;

void heapify(int a[], int n, int i) {
    int largest = i;
    int l = 2 * i + 1;
    int r = 2 * i + 2;

    if (l < n && a[l] > a[largest])
        largest = l;
    if (r < n && a[r] > a[largest])
        largest = r;

    if (largest != i) {
        swap(a[i], a[largest]);
        heapify(a, n, largest);
    }
}

void heapSort(int a[], int n) {
    for (int i = n / 2 - 1; i >= 0; i--)
        heapify(a, n, i);

    for (int i = n - 1; i > 0; i--) {
        swap(a[0], a[i]);
        heapify(a, i, 0);
    }
}

int main() {
    int a[] = {4, 10, 3, 5, 1};
    int n = 5;

    heapSort(a, n);

    for (int i = 0; i < n; i++)
        cout << a[i] << " ";

    return 0;
}

Singly Linked List

Creates nodes, inserts at the end, and displays the list.

Insert at end: O(n) per insertionSpace: O(n) for nodes
C++ source code
#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* next;
};

Node* head = nullptr;

void insert(int value) {
    Node* newNode = new Node{value, nullptr};

    if (head == nullptr) {
        head = newNode;
        return;
    }

    Node* temp = head;
    while (temp->next != nullptr)
        temp = temp->next;

    temp->next = newNode;
}

void display() {
    Node* temp = head;

    while (temp != nullptr) {
        cout << temp->data << " -> ";
        temp = temp->next;
    }

    cout << "NULL";
}

int main() {
    insert(10);
    insert(20);
    insert(30);

    display();
    return 0;
}

Sample output

10 -> 20 -> 30 -> NULL

Infix to Postfix Using Stack

Uses operator precedence and a stack to convert an expression.

Time: O(n)Space: O(n)
C++ source code
#include <iostream>
#include <stack>
#include <string>
#include <cctype>
using namespace std;

int precedence(char op) {
    if (op == '^') return 3;
    if (op == '*' || op == '/') return 2;
    if (op == '+' || op == '-') return 1;
    return 0;
}

int main() {
    string infix, postfix = "";
    stack<char> st;

    cout << "Enter infix expression: ";
    cin >> infix;

    for (char c : infix) {
        if (isalnum(static_cast<unsigned char>(c))) {
            postfix += c;
        } else if (c == '(') {
            st.push(c);
        } else if (c == ')') {
            while (!st.empty() && st.top() != '(') {
                postfix += st.top();
                st.pop();
            }
            if (!st.empty()) st.pop();
        } else {
            while (!st.empty() && st.top() != '(' &&
                   (precedence(st.top()) > precedence(c) ||
                   (precedence(st.top()) == precedence(c) && c != '^'))) {
                postfix += st.top();
                st.pop();
            }
            st.push(c);
        }
    }

    while (!st.empty()) {
        postfix += st.top();
        st.pop();
    }

    cout << "Postfix: " << postfix;
    return 0;
}

Sample input and output

Enter infix expression: A+B*C
Postfix: ABC*+

This simple version supports single-character operands and expressions without spaces.

Factorial Using Recursion

The function calls itself until it reaches the base case.

Time: O(n)Auxiliary space: O(n)
C++ source code
#include <iostream>
using namespace std;

int factorial(int n) {
    if (n <= 1)
        return 1;

    return n * factorial(n - 1);
}

int main() {
    int n;
    cout << "Enter a number: ";
    cin >> n;

    if (n < 0) {
        cout << "Factorial is not defined for negative numbers";
    } else if (n > 12) {
        cout << "Number too large for this program";
    } else {
        cout << "Factorial = " << factorial(n);
    }

    return 0;
}

Sample input and output

Enter a number: 5
Factorial = 120

Stack Using Array

Implements Push, Pop, and Display operations. Stack follows LIFO.

Push / Pop: O(1)Storage: O(n)
C++ source code
#include <iostream>
using namespace std;

const int SIZE = 5;
int st[SIZE];
int top = -1;

void push(int x) {
    if (top == SIZE - 1)
        cout << "Stack Overflow\n";
    else {
        st[++top] = x;
        cout << "Element pushed\n";
    }
}

void pop() {
    if (top == -1)
        cout << "Stack Underflow\n";
    else
        cout << "Popped: " << st[top--] << "\n";
}

void display() {
    if (top == -1) {
        cout << "Stack is empty\n";
        return;
    }

    for (int i = top; i >= 0; i--)
        cout << st[i] << " ";
    cout << "\n";
}

int main() {
    int choice, value;

    do {
        cout << "\n1.Push  2.Pop  3.Display  4.Exit\n";
        cout << "Enter choice: ";
        cin >> choice;

        switch (choice) {
            case 1:
                cout << "Enter value: ";
                cin >> value;
                push(value);
                break;
            case 2: pop(); break;
            case 3: display(); break;
            case 4: cout << "Exiting\n"; break;
            default: cout << "Invalid choice\n";
        }
    } while (choice != 4);

    return 0;
}

Example

Push 20
Display: 20 10

Binary Search Tree (BST)

Implements insertion, searching, deletion, and inorder traversal.

Average operation time: O(log n)Worst case: O(n)Storage: O(n)
C++ source code
#include <iostream>
using namespace std;

struct Node {
    int data;
    Node *left, *right;

    Node(int value) {
        data = value;
        left = right = nullptr;
    }
};

Node* insert(Node* root, int x) {
    if (root == nullptr)
        return new Node(x);

    if (x < root->data)
        root->left = insert(root->left, x);
    else if (x > root->data)
        root->right = insert(root->right, x);

    return root;
}

bool search(Node* root, int x) {
    if (root == nullptr) return false;
    if (root->data == x) return true;
    if (x < root->data) return search(root->left, x);
    return search(root->right, x);
}

Node* minNode(Node* root) {
    while (root && root->left)
        root = root->left;
    return root;
}

Node* deleteNode(Node* root, int x) {
    if (root == nullptr) return nullptr;

    if (x < root->data)
        root->left = deleteNode(root->left, x);
    else if (x > root->data)
        root->right = deleteNode(root->right, x);
    else {
        if (root->left == nullptr) {
            Node* temp = root->right;
            delete root;
            return temp;
        }
        if (root->right == nullptr) {
            Node* temp = root->left;
            delete root;
            return temp;
        }

        Node* temp = minNode(root->right);
        root->data = temp->data;
        root->right = deleteNode(root->right, temp->data);
    }

    return root;
}

void inorder(Node* root) {
    if (root) {
        inorder(root->left);
        cout << root->data << " ";
        inorder(root->right);
    }
}

int main() {
    Node* root = nullptr;
    root = insert(root, 50);
    root = insert(root, 30);
    root = insert(root, 70);
    root = insert(root, 20);
    root = insert(root, 40);

    cout << "BST: ";
    inorder(root);

    cout << "\nSearch 40: "
         << (search(root, 40) ? "Found" : "Not Found");

    root = deleteNode(root, 30);
    cout << "\nAfter deletion: ";
    inorder(root);

    return 0;
}

Sample output

BST: 20 30 40 50 70
Search 40: Found
After deletion: 20 40 50 70

Duplicate values are ignored in this implementation. Inorder traversal displays the keys in ascending order.

Dijkstra's Shortest Path Algorithm

Finds shortest distances from one source vertex to all other vertices in a graph with non-negative edge weights.

Time: O(V²)Auxiliary space: O(V)Graph matrix storage: O(V²)
C++ source code
#include <iostream>
#include <climits>
using namespace std;

const int V = 5;

int minDistance(int dist[], bool visited[]) {
    int minValue = INT_MAX, index = -1;

    for (int i = 0; i < V; i++) {
        if (!visited[i] && dist[i] < minValue) {
            minValue = dist[i];
            index = i;
        }
    }

    return index;
}

void dijkstra(int graph[V][V], int src) {
    int dist[V];
    bool visited[V] = {false};

    for (int i = 0; i < V; i++)
        dist[i] = INT_MAX;

    dist[src] = 0;

    for (int count = 0; count < V; count++) {
        int u = minDistance(dist, visited);
        if (u == -1) break;

        visited[u] = true;

        for (int v = 0; v < V; v++) {
            if (!visited[v] && graph[u][v] > 0 &&
                dist[u] != INT_MAX &&
                dist[u] + graph[u][v] < dist[v]) {
                dist[v] = dist[u] + graph[u][v];
            }
        }
    }

    cout << "Vertex  Distance from source\n";
    for (int i = 0; i < V; i++) {
        cout << i << "       ";
        if (dist[i] == INT_MAX)
            cout << "INF";
        else
            cout << dist[i];
        cout << "\n";
    }
}

int main() {
    int graph[V][V] = {
        {0, 10, 0, 0, 5},
        {10, 0, 1, 0, 2},
        {0, 1, 0, 4, 0},
        {0, 0, 4, 0, 3},
        {5, 2, 0, 3, 0}
    };

    dijkstra(graph, 0);
    return 0;
}

Sample output

Vertex  Distance from source
0       0
1       7
2       8
3       8
4       5

Vertex 0 is the source. This adjacency-matrix version uses 0 to mean “no edge,” so it does not represent zero-weight edges between distinct vertices.
